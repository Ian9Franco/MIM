### 1. El Proceso Main (`main.ts` / `index.ts`)

Este es el cerebro de tu app standalone. Se encarga de leer/escribir el JSON local y de orquestar la ventana invisible que va a scrapear CurseForge cuando la caché expire.

```typescript
import { app, BrowserWindow, ipcMain } from 'electron';
import * as path from 'path';
import * as fs from 'fs';

const CACHE_PATH = path.join(app.getPath('userData'), 'cache_colecciones.json');
const CACHE_EXPIRATION_DAYS = 7;

// 1. Función para leer la caché local
ipcMain.handle('get-cached-collections', async () => {
  if (!fs.existsSync(CACHE_PATH)) return null;
  try {
    const data = fs.readFileSync(CACHE_PATH, 'utf-8');
    return JSON.parse(data);
  } catch (error) {
    console.error("Error leyendo la caché:", error);
    return null;
  }
});

// 2. Orquestador del Scraping en segundo plano
ipcMain.handle('trigger-background-scraping', async () => {
  return new Promise((resolve) => {
    // Creamos una ventana COMPLETAMENTE INVISIBLE
    const scraperWindow = new BrowserWindow({
      show: false,
      webPreferences: {
        images: false, // Desactivamos imágenes para que cargue volando
        nodeIntegration: false,
        contextIsolation: true
      }
    });

    // Cargamos la URL del HUB de colecciones
    scraperWindow.loadURL('https://www.curseforge.com/community-picks/minecraft');

    scraperWindow.webContents.on('did-finish-load', async () => {
      try {
        // Inyectamos JS en la página para extraer los nombres y slugs de las tarjetas
        const colecciones = await scraperWindow.webContents.executeJavaScript(`
          (() => {
            // Buscamos todos los links que dicen "View Community Picks"
            const links = Array.from(document.querySelectorAll('a')).filter(a => a.innerText.includes('View Community Picks'));
            
            return links.map(link => {
              // Subimos al contenedor de la tarjeta para buscar el título y la imagen
              const card = link.closest('.flex') || link.parentElement; // Ajustar según DOM real
              const titulo = card?.querySelector('h3')?.innerText || 'Colección sin título';
              const imgUrl = card?.querySelector('img')?.src || '';
              
              return {
                id: link.getAttribute('href').split('/').pop(), // Extrae el slug final
                titulo,
                slug: link.getAttribute('href'),
                thumbnail: imgUrl
              };
            });
          })();
        `);

        // TODO: Acá podrías iterar por cada slug si necesitás entrar a sacar los IDs de los mods.
        // Por ahora, guardamos la lista del HUB con el timestamp actual.
        const cacheData = {
          ultima_actualizacion: new Date().toISOString(),
          colecciones
        };

        fs.writeFileSync(CACHE_PATH, JSON.stringify(cacheData, null, 2), 'utf-8');
        
        scraperWindow.close();
        resolve({ success: true, data: cacheData });
      } catch (err) {
        console.error("Falló el scraping local:", err);
        scraperWindow.close();
        resolve({ success: false, error: err });
      }
    });
  });
});

```

        setData(cache.colecciones);
        setLoading(false);
        
        // 2. Verificar si pasaron más de 7 días
        const ultimaVez = new Date(cache.ultima_actualizacion).getTime();
        const ahora = new Date().getTime();
        const diasPasados = (ahora - ultimaVez) / (1000 * 60 * 60 * 24);
